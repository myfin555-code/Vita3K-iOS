cmake_minimum_required(VERSION 3.22)
project(Vita3KiOSNativeSupportTests LANGUAGES CXX)
enable_testing()
add_executable(ime_text_tests IMETextTests.cpp)
target_compile_features(ime_text_tests PRIVATE cxx_std_17)
target_include_directories(ime_text_tests PRIVATE ../../vita3k/ime/include)
add_test(NAME ime_text_tests COMMAND ime_text_tests)

add_executable(guest_range_tests GuestRangeTests.cpp)
target_compile_features(guest_range_tests PRIVATE cxx_std_17)
target_include_directories(guest_range_tests PRIVATE ../../vita3k/mem/include)
add_test(NAME guest_range_tests COMMAND guest_range_tests)

find_package(Threads REQUIRED)
add_executable(runtime_budget_tests RuntimeBudgetTests.cpp)
target_compile_features(runtime_budget_tests PRIVATE cxx_std_17)
target_include_directories(runtime_budget_tests PRIVATE ../../vita3k/cpu/include ../../vita3k/mem/include)
target_link_libraries(runtime_budget_tests PRIVATE Threads::Threads)
add_test(NAME runtime_budget_tests COMMAND runtime_budget_tests)
set_tests_properties(runtime_budget_tests PROPERTIES TIMEOUT 30)
